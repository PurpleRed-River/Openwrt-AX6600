# =============================================
# RivWRT 增量配置
#
# 叠加在 Config/GENERAL_AX6600.txt（通用基座）之上，
# 由 WRT-CORE.yml 按 GENERAL_AX6600_${PROFILE}.txt 规则自动拼接，
# 后写的条目按 kconfig 规则覆盖基座同名项。
#
# 设备：JDCloud RE-CS-02（京东云 AX6600 雅典娜）
# 平台：Qualcomm IPQ6010 / qualcommax-ipq60xx
# 适配分区：12M 内核 + 2G rootfs（chenxin527/uboot-qsdk12.5-build）
# =============================================
# BPF 工具链：dae 的 eBPF 代码用宿主 clang 编译（需 clang>=12，runner 预装满足）。
# 注意：BPF_TOOLCHAIN choice 的可见性挂在 "if DEVEL" 上，基座 CONFIG_DEVEL=n
# 会导致 HOST 行被 kconfig 静默丢弃（落到空选择 → /invalid/clang 报错），
# 故必须同时开启 DEVEL；DEVEL 仅控制开发者菜单可见性，无实质副作用。
# 内核 BTF：开启。已实测设备 GPT：A 槽 0:HLOS=12288KB(12MiB)+rootfs=2GiB，
# KERNEL_SIZE 已在 Settings.sh 适配为 12288k，内核+BTF 段约 7-8MiB 余量充足。
# 开启后 /sys/kernel/btf/vmlinux 原生可用，dae/bandix/未来 eBPF 工具全覆盖；
# vmlinux-btf 独立包仍作为 dae 硬依赖编入兜底（dwarves/pahole 已装可构建）。
# =============================================
CONFIG_DEVEL=y
CONFIG_BPF_TOOLCHAIN_HOST=y
CONFIG_DEBUG_INFO_BTF=y

# =============================================

# -----------------
# LED 点阵屏控制
# 使用上游树内官方版 package/emortal/luci-app-athena-led（ones20250 同款，
# 自带预编译二进制零下载）；unraveloop 增强版 release 资产已 404 故不采用
# -----------------
CONFIG_PACKAGE_luci-app-athena-led=y

# -----------------
# LuCI 主题
# 来源：eamonxg/luci-theme-aurora（Vite + Tailwind 现代 UI，深浅色）
# 默认主题替换由 Settings.sh 完成（luci collection 依赖 bootstrap → aurora）
# -----------------
CONFIG_PACKAGE_luci-theme-aurora=y
# 主题设置界面（aurora 官方配套：预设/色彩/布局/背景）
CONFIG_PACKAGE_luci-app-aurora-config=y

# -----------------
# 流量行为统计（bandix-plus：eBPF 旁路观察）
# 定位：纯观察，不参与转发路径，不影响 NSS 加速性能。
# 注意：被 NSS 硬件加速的直连流量不进内核统计路径，bandix 数字会偏低；
# 需要看精确报表时在 LuCI 停用 NSS ecm（流量回落内核即数全），看完再开。
# 依赖：bandix-plus（后端，同仓库族 timsaya/openwrt-bandix-plus）+ eBPF 内核
# -----------------
CONFIG_PACKAGE_bandix-plus=y
CONFIG_PACKAGE_luci-app-bandix-plus=y

# -----------------
# 透明代理（kenzok8/openwrt-daede：dae eBPF 内核 + luci-app-daede 管理界面）
# 分工：dae 规则判定代理的流量由 dae 接管（不经 NSS）；直连流量满血 NSS 加速。
# DNS 内外网分流/防污染由 dae 内置 DNS 模块承担（geosite/geoip），
# dnsmasq 仅保留 DHCP 与内网 split-horizon 职责，不装 mosdns/smartdns。
# luci-app-daede 默认经 kconfig choice 拉入 daed 后端；dae/daed 同一套 UCI 可切换。
# -----------------
CONFIG_PACKAGE_dae=y
CONFIG_PACKAGE_luci-app-daede=y

# -----------------
# Podman 容器环境（连带 veth 内核支持，覆盖基座 kmod-veth=n）
# 只装环境，不默认跑任何容器；容器数据建议落 eMMC 独立分区
# -----------------
CONFIG_PACKAGE_podman=y
# compose 编排（CLI 使用：podman-compose up -d）
CONFIG_PACKAGE_podman-compose=y
CONFIG_PACKAGE_kmod-veth=y

# -----------------
# IPv6（基座 ImmortalWrt 默认已含，显式声明保证开启）
# -----------------
CONFIG_PACKAGE_odhcp6c=y
CONFIG_PACKAGE_odhcpd-ipv6only=y

# -----------------
# fullcone NAT（游戏机/P2P 更友好的 NAT 行为，nftables 实现）
# -----------------
CONFIG_PACKAGE_kmod-nft-fullcone=y

# -----------------
# 接口级流量总量统计（基于网卡计数器，NSS 加速流量照数，永远准确；
# 与 bandix 互补：vnstat 看总量趋势，bandix 看设备明细）
# -----------------
CONFIG_PACKAGE_vnstat=y
CONFIG_PACKAGE_luci-app-vnstat=y

# 状态历史曲线（CPU/内存/接口流量图表，collectd 常驻约 10MB）
CONFIG_PACKAGE_luci-app-statistics=y

# NSS 加速管理页（自建包：开关 + 每 Core 负载）
CONFIG_PACKAGE_luci-app-rivwrt-nss=y

# -----------------
# 文件共享：ksmbd（内核态 SMB，替换基座的 Samba4）
# 1G RAM 设备省 ~34MB 常驻（smbd/nmbd/notifyd/cleanupd 全家），
# eMMC 大文件共享吞吐更高；代价：无打印共享/域控、macOS Time Machine 支持有限
# 基座 GENERAL_AX6600.txt 中的 luci-app-samba4=y 在此覆盖关闭
# -----------------
CONFIG_PACKAGE_luci-app-samba4=n
CONFIG_PACKAGE_samba4-server=n
CONFIG_PACKAGE_ksmbd-server=y
CONFIG_PACKAGE_luci-app-ksmbd=y

# -----------------
# 网络唤醒补充：上游 wolplus 已换代为 wolultra（基座的 luci-app-wolplus=y 成死行），
# 此处启用新包，功能不变（etherwake 内核，LuCI 界面唤醒）
# -----------------
CONFIG_PACKAGE_luci-app-wolultra=y
